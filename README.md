    // Chọn nhanh dịch vụ đưa xuống Form đặt lịch khám
    function selectServiceInForm(serviceName) {
      const selectEl = document.getElementById('serviceSelect');
      const symptomsEl = document.getElementById('symptomsText');
      if (selectEl) {
        let matched = false;
        for (let i = 0; i < selectEl.options.length; i++) {
          if (selectEl.options[i].value.toLowerCase().includes(serviceName.toLowerCase())) {
            selectEl.selectedIndex = i;
            matched = true;
            break;
          }
        }
        if (!matched && symptomsEl && !symptomsEl.value.includes(serviceName)) {
          symptomsEl.value = 'Đăng ký: ' + serviceName + (symptomsEl.value ? ' - ' + symptomsEl.value : '');
        }
      }
    }
